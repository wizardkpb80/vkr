&НаСервере
Функция ПолучитьПрогнозПоИд(ИдЗаказа, ТипПрогноза)
    // ТипПрогноза: "qwen" или "baseline"
    АдресСервера = "http://192.168.152.57:8000/1c/";
    Если ТипПрогноза = "qwen" Тогда
        Адрес = АдресСервера + "predict_qwen_by_order_id";
    Иначе
        Адрес = АдресСервера + "predict_baseline_by_order_id";
    КонецЕсли;
    
    HTTPСоединение = Новый HTTPСоединение(Адрес);
    HTTPЗапрос = Новый HTTPЗапрос();
    HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json");
    ТелоЗапроса = "{""order_id"":""" + Строка(ИдЗаказа) + """}";
    HTTPЗапрос.УстановитьТелоИзСтроки(ТелоЗапроса);
    
    Попытка
        Ответ = HTTPСоединение.ОтправитьДляОбработки(HTTPЗапрос);
        Если Ответ.КодСостояния = 200 Тогда
            ЧтениеJSON = Новый ЧтениеJSON(Ответ.ПолучитьТелоКакСтроку());
            Результат = ПрочитатьJSON(ЧтениеJSON, Ложь);
            Возврат Результат.predicted_duration_minutes;
        Иначе
            Сообщить("Ошибка HTTP (" + ТипПрогноза + "): " + Ответ.КодСостояния);
            Возврат Неопределено;
        КонецЕсли;
    Исключение
        Сообщить("Ошибка соединения (" + ТипПрогноза + "): " + ОписаниеОшибки());
        Возврат Неопределено;
    КонецПопытки;
КонецФункции

&НаСервере
Процедура ОбновитьПрогнозы()
    СсылкаНаЗаказ = ЭтаФорма.Объект.Ссылка;
    
    // Qwen – если есть картинка
    Если ЗначениеЗаполнено(ЭтаФорма.Объект.ФайлКартинки) Тогда
        Прогноз = ПолучитьПрогнозПоИд(СсылкаНаЗаказ, "qwen");
        Если Прогноз <> Неопределено Тогда
            ЭтаФорма.ПрогнозQwen = Прогноз;
        Иначе
            ЭтаФорма.ПрогнозQwen = Неопределено;
        КонецЕсли;
    Иначе
        ЭтаФорма.ПрогнозQwen = Неопределено;
    КонецЕсли;
    
    // Baseline – если есть строки в табличной части Запасы
    Если ЭтаФорма.Объект.Запасы.Количество() > 0 Тогда
        Прогноз = ПолучитьПрогнозПоИд(СсылкаНаЗаказ, "baseline");
        Если Прогноз <> Неопределено Тогда
            ЭтаФорма.ПрогнозBaseline = Прогноз;
        Иначе
            ЭтаФорма.ПрогнозBaseline = Неопределено;
        КонецЕсли;
    Иначе
        ЭтаФорма.ПрогнозBaseline = Неопределено;
    КонецЕсли;
КонецПроцедуры

&НаКлиенте
Процедура КомандаРассчитатьПрогноз(Команда)
    ОбновитьПрогнозыНаСервере();
КонецПроцедуры

&НаСервере
Процедура ОбновитьПрогнозыНаСервере()
    ОбновитьПрогнозы();
КонецПроцедуры

&НаСервере
Процедура ПриСозданииНаСервере(Отказ, СтандартнаяОбработка)
    ОбновитьПрогнозы();
